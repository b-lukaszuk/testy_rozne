*** 1.1.4 Compound Procedures

#+begin_src

square :: Num a => a -> a
square x = x * x

square 21
-- 441

square $ 2 + 5
-- 49

square $ square 3
-- 81

#+end_src

#+begin_src

sumOfSquares :: Num a => a -> a -> a
sumOfSquares x y = square x + square y

sumOfSquares 3 4
-- 25

#+end_src
