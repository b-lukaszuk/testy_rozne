*** 1.1.7 Example: Square Roots by Newton's Method

$\sqrt{x}$ = the y such that $y \ge  0$ and $y^2 = x$

#+begin_src

getAvg :: Fractional a => a -> a -> a
getAvg x y = (x + y) / 2

getNewGuessForSqrtOfX :: Fractional a => a -> a -> a
getNewGuessForSqrtOfX oldGuess x = getAvg oldGuess $ x / oldGuess

getAbsDiff :: Num a => a -> a -> a
getAbsDiff x y = abs $ x - y

areApproxEqual :: (Ord a, Fractional a) => a -> a -> Bool
areApproxEqual x y = getAbsDiff x y < 0.001

square :: Num a => a -> a
square x = x * x

sqrtIter' :: (Ord a, Fractional a) => a -> a -> a
sqrtIter' guess x
  | areApproxEqual (square guess) x = guess
  | otherwise = sqrtIter' (getNewGuessForSqrtOfX guess x) x

sqrtIter :: (Ord a, Fractional a) => a -> a
sqrtIter x = sqrtIter' 1 x

#+end_src

tests:

#+begin_src

λ> sqrtIter 9
3.00009155413138

λ> sqrtIter $ 100 + 37
11.704699917758145

λ> sqrtIter $ sqrtIter 2 + sqrtIter 3
1.7739279023207892

λ> square (sqrtIter 1000)
1000.000369924366

#+end_src

**** Ex. 1.8. Newton's method for cube roots

Implement Newton's method for finding cube roots.

Formula for finding a new, better guess:

$$z = \frac{x/y^{2}+2y}{3}$$

where:

z - new guess

x - value for which cube root is to be calculated ($\sqrt[3]{x}$ )

y - old guess

#+begin_src

pow' :: (Ord a, Num a, Num b) => b -> a -> b -> b
pow' base exponent acc
  | exponent < 1 = acc
  | otherwise = pow' base (exponent - 1) (acc * base)

pow :: (Num a, Ord b, Num b) => a -> b -> a
pow base exponent = pow' base exponent 1

getNewGuess :: Fractional a => a -> a -> a
getNewGuess oldGuess x = (x / (pow oldGuess 2) + (2 * oldGuess)) / 3

getAbsDiff :: Num a => a -> a -> a
getAbsDiff x y = abs $ x - y

areApproxEqual :: (Ord a, Fractional a) => a -> a -> Bool
areApproxEqual x y = getAbsDiff x y < 0.001

cbrtIter' :: (Ord a, Fractional a) => a -> a -> a
cbrtIter' guess x = if areApproxEqual (pow guess 3) x
  then guess
  else cbrtIter' (getNewGuess guess x) x

cbrtIter :: (Ord a, Fractional a) => a -> a
cbrtIter x = cbrtIter' 1 x
