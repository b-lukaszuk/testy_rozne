*** 1.1.4 Compound Procedures

#+begin_src

square :: Num a => a -> a
square x = x * x

square 21
-- 441

square $ 2 + 5
-- 49

square $ square 3
-- 81

#+end_src

#+begin_src

sumOfSquares :: Num a => a -> a -> a
sumOfSquares x y = square x + square y

sumOfSquares 3 4
-- 25

#+end_src

*** 1.1.6 Conditional Expressions and Predicates

#+begin_src

myAbs :: (Ord a, Num a) => a -> a
myAbs x
  | x > 0 = x
  | x == 0 = 0
  | otherwise = -x

#+end_src

or

#+begin_src

myAbs' :: (Ord a, Num a) => a -> a
myAbs' x
  | x < 0 = -x
  | otherwise = x

#+end_src

or

#+begin_src

myAbs'' :: (Ord a, Num a) => a -> a
myAbs'' x = if x < 0 then -x else x

#+end_src
